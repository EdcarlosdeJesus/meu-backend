const express = require('express');
const app = express();
const port = 8080;

app.get('/', (req, res) => {
  res.send('Backend simples rodando no Render!');
});

app.get('/api/test', (req, res) => {
  res.json({ message: 'API funcionando!' });
});

app.listen(port, () => {
  console.log(`Servidor rodando na porta ${port}`);
});
